import React, { useState } from 'react';
import { useCart } from '../context/CartContext';
import { api } from '../services/api';
import {
  X,
  MapPin,
  Phone,
  User,
  FileText,
  CreditCard,
  Banknote,
  Store,
  QrCode,
  Copy,
  Check,
  PhoneCall,
  ShieldCheck,
  AlertCircle
} from 'lucide-react';

interface CheckoutModalProps {
  onOrderSuccess: (orderId: string) => void;
}

export const CheckoutModal: React.FC<CheckoutModalProps> = ({ onOrderSuccess }) => {
  const {
    isCheckoutOpen,
    setIsCheckoutOpen,
    cart,
    subtotal,
    deliveryFee,
    clearCart,
    settings,
    setLastPlacedOrder
  } = useCart();

  const [orderType, setOrderType] = useState<'delivery' | 'takeaway'>('delivery');
  const [customerName, setCustomerName] = useState('');
  const [customerPhone, setCustomerPhone] = useState('');
  const [address, setAddress] = useState('');
  const [landmark, setLandmark] = useState('');
  const [specialInstructions, setSpecialInstructions] = useState('');
  const [paymentMethod, setPaymentMethod] = useState<'cod' | 'upi' | 'restaurant'>('cod');
  const [upiTxnId, setUpiTxnId] = useState('');
  const [copiedUpi, setCopiedUpi] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [errorMessage, setErrorMessage] = useState('');

  if (!isCheckoutOpen) return null;

  const effectiveDeliveryFee = orderType === 'takeaway' ? 0 : deliveryFee;
  const effectiveTotal = subtotal + effectiveDeliveryFee;

  const handleCopyUPI = () => {
    navigator.clipboard.writeText(settings.upiId);
    setCopiedUpi(true);
    setTimeout(() => setCopiedUpi(false), 2000);
  };

  const validateForm = () => {
    if (!customerName.trim()) {
      setErrorMessage('Please enter your full name');
      return false;
    }
    const cleanPhone = customerPhone.replace(/[^0-9]/g, '');
    if (cleanPhone.length < 10) {
      setErrorMessage('Please enter a valid 10-digit mobile number');
      return false;
    }
    if (orderType === 'delivery' && !address.trim()) {
      setErrorMessage('Please enter your delivery address in Sasaram');
      return false;
    }
    setErrorMessage('');
    return true;
  };

  const generateWhatsAppMessage = () => {
    const itemsText = cart
      .map(
        (ci, i) =>
          `  ${i + 1}. *${ci.item.name}* x ${ci.quantity} = ₹${ci.item.price * ci.quantity}`
      )
      .join('\n');

    let msg = `*AL ISHRA RESTAURANT - ONLINE ORDER*\n`;
    msg += `━━━━━━━━━━━━━━━━━━━━\n`;
    msg += `👤 *Customer:* ${customerName.trim()}\n`;
    msg += `📞 *Phone:* ${customerPhone.trim()}\n`;
    msg += `🛵 *Order Type:* ${orderType === 'delivery' ? 'Home Delivery' : 'Restaurant Takeaway/Pickup'}\n`;

    if (orderType === 'delivery') {
      msg += `📍 *Delivery Address:* ${address.trim()}\n`;
      if (landmark.trim()) msg += `🏷️ *Landmark:* ${landmark.trim()}\n`;
    }

    if (specialInstructions.trim()) msg += `📝 *Instructions:* ${specialInstructions.trim()}\n`;

    msg += `━━━━━━━━━━━━━━━━━━━━\n`;
    msg += `*SELECTED FOOD ITEMS:*\n${itemsText}\n`;
    msg += `━━━━━━━━━━━━━━━━━━━━\n`;
    msg += `• *Subtotal:* ₹${subtotal}\n`;
    msg += `• *Delivery Fee:* ₹${effectiveDeliveryFee}\n`;
    msg += `💰 *TOTAL BILL:* ₹${effectiveTotal}\n`;
    msg += `💳 *Payment Method:* ${
      paymentMethod === 'cod'
        ? 'Cash on Delivery (COD)'
        : paymentMethod === 'upi'
        ? `UPI (Txn ID: ${upiTxnId || 'Direct UPI Transfer'})`
        : 'Pay at Restaurant'
    }\n`;
    msg += `━━━━━━━━━━━━━━━━━━━━\n`;
    msg += `Please confirm my order and start preparation!`;

    return msg;
  };

  const handlePlaceOrder = async (isWhatsApp = false) => {
    if (!validateForm()) return;
    setIsSubmitting(true);

    try {
      const orderItems = cart.map(ci => ({
        id: ci.item.id,
        name: ci.item.name,
        quantity: ci.quantity,
        price: ci.item.price,
        isCombo: ci.isCombo
      }));

      const newOrder = await api.createOrder({
        customerName: customerName.trim(),
        customerPhone: customerPhone.trim(),
        orderType,
        address: orderType === 'delivery' ? address.trim() : undefined,
        landmark: orderType === 'delivery' ? landmark.trim() : undefined,
        specialInstructions: specialInstructions.trim() || undefined,
        paymentMethod,
        items: orderItems,
        subtotal,
        deliveryFee: effectiveDeliveryFee,
        total: effectiveTotal,
        upiTxnId: paymentMethod === 'upi' ? upiTxnId.trim() : undefined
      });

      setLastPlacedOrder(newOrder);
      clearCart();
      setIsCheckoutOpen(false);

      if (isWhatsApp) {
        const rawNum = settings.whatsappNumber.replace(/[^0-9]/g, '');
        const num = rawNum.startsWith('91') ? rawNum : `91${rawNum}`;
        const waText = encodeURIComponent(generateWhatsAppMessage());
        window.open(`https://wa.me/${num}?text=${waText}`, '_blank');
      }

      onOrderSuccess(newOrder.id);
    } catch (err) {
      console.error('Failed to place order:', err);
      setErrorMessage('Could not place order. Please try again or order on WhatsApp.');
    } finally {
      setIsSubmitting(false);
    }
  };

  const upiIntentString = `upi://pay?pa=${settings.upiId}&pn=${encodeURIComponent(
    settings.upiName
  )}&am=${effectiveTotal}&cu=INR&tn=AlIshraOrder`;
  const qrCodeUrl = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(
    upiIntentString
  )}`;

  return (
    <div className="fixed inset-0 z-50 overflow-y-auto bg-black/85 backdrop-blur-md flex items-center justify-center p-4">
      <div className="relative w-full max-w-2xl bg-[#141414] border border-[#2E2E2E] rounded-3xl shadow-2xl overflow-hidden my-6">
        <div className="p-5 sm:p-6 border-b border-[#242424] flex items-center justify-between bg-[#181818]">
          <div>
            <span className="text-[11px] font-bold tracking-widest text-[#D4AF37] uppercase block font-cinzel">
              AL ISHRA RESTAURANT
            </span>
            <h3 className="text-xl font-bold text-white font-cinzel">Complete Your Order</h3>
          </div>
          <button
            onClick={() => setIsCheckoutOpen(false)}
            className="p-2 text-gray-400 hover:text-white rounded-lg hover:bg-[#252525]"
            aria-label="Close checkout"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {errorMessage && (
          <div className="mx-6 mt-4 p-3 bg-red-950/80 border border-red-500/50 rounded-xl flex items-center gap-2 text-xs text-red-200">
            <AlertCircle className="w-4 h-4 text-red-400 shrink-0" />
            <span>{errorMessage}</span>
          </div>
        )}

        <div className="p-6 space-y-6 max-h-[75vh] overflow-y-auto">
          <div>
            <label className="block text-xs font-bold text-gray-400 uppercase tracking-wider mb-2">
              Select Order Type
            </label>
            <div className="grid grid-cols-2 gap-3">
              <button
                type="button"
                onClick={() => setOrderType('delivery')}
                className={`p-3.5 rounded-xl border flex items-center justify-center gap-2.5 font-bold text-sm transition-all cursor-pointer ${
                  orderType === 'delivery'
                    ? 'bg-[#D4AF37]/15 border-[#D4AF37] text-[#F5E5A4]'
                    : 'bg-[#1a1a1a] border-[#292929] text-gray-400 hover:text-white'
                }`}
              >
                <MapPin className="w-4 h-4 text-[#D4AF37]" />
                <span>Home Delivery</span>
              </button>

              <button
                type="button"
                onClick={() => setOrderType('takeaway')}
                className={`p-3.5 rounded-xl border flex items-center justify-center gap-2.5 font-bold text-sm transition-all cursor-pointer ${
                  orderType === 'takeaway'
                    ? 'bg-[#D4AF37]/15 border-[#D4AF37] text-[#F5E5A4]'
                    : 'bg-[#1a1a1a] border-[#292929] text-gray-400 hover:text-white'
                }`}
              >
                <Store className="w-4 h-4 text-[#D4AF37]" />
                <span>Takeaway / Pickup</span>
              </button>
            </div>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div>
              <label className="block text-xs font-medium text-gray-300 mb-1.5 flex items-center gap-1.5">
                <User className="w-3.5 h-3.5 text-[#D4AF37]" />
                <span>Full Name *</span>
              </label>
              <input
                type="text"
                value={customerName}
                onChange={e => setCustomerName(e.target.value)}
                placeholder="e.g. Mohd Rashid"
                className="w-full bg-[#1a1a1a] border border-[#2b2b2b] rounded-xl px-3.5 py-2.5 text-sm text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
                required
              />
            </div>

            <div>
              <label className="block text-xs font-medium text-gray-300 mb-1.5 flex items-center gap-1.5">
                <Phone className="w-3.5 h-3.5 text-[#D4AF37]" />
                <span>Mobile Number *</span>
              </label>
              <div className="relative">
                <span className="absolute left-3.5 top-1/2 -translate-y-1/2 text-xs text-gray-400 font-semibold">
                  +91
                </span>
                <input
                  type="tel"
                  value={customerPhone}
                  onChange={e => setCustomerPhone(e.target.value.replace(/[^0-9]/g, '').slice(0, 10))}
                  placeholder="98765 43210"
                  className="w-full bg-[#1a1a1a] border border-[#2b2b2b] rounded-xl pl-12 pr-3.5 py-2.5 text-sm text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
                  required
                />
              </div>
            </div>
          </div>

          {orderType === 'delivery' && (
            <div className="space-y-4 pt-1">
              <div>
                <label className="block text-xs font-medium text-gray-300 mb-1.5 flex items-center gap-1.5">
                  <MapPin className="w-3.5 h-3.5 text-[#D4AF37]" />
                  <span>Delivery Address (in Sasaram) *</span>
                </label>
                <textarea
                  rows={2}
                  value={address}
                  onChange={e => setAddress(e.target.value)}
                  placeholder="Colony / Street, House No., Area in Sasaram, Rohtas"
                  className="w-full bg-[#1a1a1a] border border-[#2b2b2b] rounded-xl p-3 text-sm text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
                  required
                />
              </div>

              <div>
                <label className="block text-xs font-medium text-gray-300 mb-1.5">
                  Famous Landmark (Optional)
                </label>
                <input
                  type="text"
                  value={landmark}
                  onChange={e => setLandmark(e.target.value)}
                  placeholder="Near Post Office Chowk / Bus Stand / Tara Chandi Gate"
                  className="w-full bg-[#1a1a1a] border border-[#2b2b2b] rounded-xl px-3.5 py-2.5 text-sm text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
                />
              </div>
            </div>
          )}

          <div>
            <label className="block text-xs font-medium text-gray-300 mb-1.5 flex items-center gap-1.5">
              <FileText className="w-3.5 h-3.5 text-[#D4AF37]" />
              <span>Cooking & Special Instructions (Optional)</span>
            </label>
            <input
              type="text"
              value={specialInstructions}
              onChange={e => setSpecialInstructions(e.target.value)}
              placeholder="e.g. Extra spicy, no onions, please send extra ketchup sachets"
              className="w-full bg-[#1a1a1a] border border-[#2b2b2b] rounded-xl px-3.5 py-2.5 text-sm text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
            />
          </div>

          <div>
            <label className="block text-xs font-bold text-gray-400 uppercase tracking-wider mb-2.5">
              Choose Payment Method
            </label>
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
              <button
                type="button"
                onClick={() => setPaymentMethod('cod')}
                className={`p-3.5 rounded-xl border flex flex-col items-center text-center gap-1.5 transition-all cursor-pointer ${
                  paymentMethod === 'cod'
                    ? 'bg-[#D4AF37]/15 border-[#D4AF37] text-white'
                    : 'bg-[#1a1a1a] border-[#292929] text-gray-400 hover:text-white'
                }`}
              >
                <Banknote className="w-5 h-5 text-[#D4AF37]" />
                <span className="text-xs font-bold">Cash on Delivery</span>
                <span className="text-[10px] text-gray-400">Pay when delivered</span>
              </button>

              <button
                type="button"
                onClick={() => setPaymentMethod('upi')}
                className={`p-3.5 rounded-xl border flex flex-col items-center text-center gap-1.5 transition-all cursor-pointer ${
                  paymentMethod === 'upi'
                    ? 'bg-[#D4AF37]/15 border-[#D4AF37] text-white'
                    : 'bg-[#1a1a1a] border-[#292929] text-gray-400 hover:text-white'
                }`}
              >
                <QrCode className="w-5 h-5 text-[#D4AF37]" />
                <span className="text-xs font-bold">Instant UPI</span>
                <span className="text-[10px] text-gray-400">GPay, PhonePe, Paytm</span>
              </button>

              <button
                type="button"
                onClick={() => setPaymentMethod('restaurant')}
                className={`p-3.5 rounded-xl border flex flex-col items-center text-center gap-1.5 transition-all cursor-pointer ${
                  paymentMethod === 'restaurant'
                    ? 'bg-[#D4AF37]/15 border-[#D4AF37] text-white'
                    : 'bg-[#1a1a1a] border-[#292929] text-gray-400 hover:text-white'
                }`}
              >
                <Store className="w-5 h-5 text-[#D4AF37]" />
                <span className="text-xs font-bold">Pay at Counter</span>
                <span className="text-[10px] text-gray-400">At restaurant pickup</span>
              </button>
            </div>
          </div>

          {paymentMethod === 'upi' && (
            <div className="p-4 rounded-2xl bg-[#1c1c1c] border border-[#D4AF37]/50 space-y-4">
              <div className="flex items-center justify-between">
                <div className="flex items-center gap-2">
                  <ShieldCheck className="w-4 h-4 text-emerald-400" />
                  <span className="text-xs font-bold text-white">Verified Restaurant UPI Details</span>
                </div>
                <span className="text-xs font-black text-[#D4AF37] font-cinzel">Amount: ₹{effectiveTotal}</span>
              </div>

              <div className="flex flex-col sm:flex-row items-center gap-4 bg-[#141414] p-3.5 rounded-xl border border-[#2b2b2b]">
                <div className="p-2 bg-white rounded-lg shrink-0">
                  <img src={qrCodeUrl} alt="UPI Payment QR Code" className="w-24 h-24" />
                </div>
                <div className="flex-1 space-y-2 text-center sm:text-left">
                  <div>
                    <span className="text-[10px] text-gray-400 uppercase tracking-wider block">Payee Name</span>
                    <span className="text-xs font-bold text-white">{settings.upiName}</span>
                  </div>
                  <div>
                    <span className="text-[10px] text-gray-400 uppercase tracking-wider block">UPI ID</span>
                    <div className="flex items-center justify-center sm:justify-start gap-2">
                      <code className="text-xs font-mono text-[#F5E5A4] bg-black/40 px-2 py-0.5 rounded">
                        {settings.upiId}
                      </code>
                      <button
                        type="button"
                        onClick={handleCopyUPI}
                        className="text-gray-400 hover:text-white text-xs flex items-center gap-1"
                        title="Copy UPI ID"
                      >
                        {copiedUpi ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
                      </button>
                    </div>
                  </div>
                  <p className="text-[11px] text-gray-400">
                    Scan via Google Pay, PhonePe, Paytm or BHIM UPI to pay ₹{effectiveTotal}.
                  </p>
                </div>
              </div>

              <div>
                <label className="block text-xs font-medium text-gray-300 mb-1">
                  UPI UTR / Transaction Reference (Optional)
                </label>
                <input
                  type="text"
                  value={upiTxnId}
                  onChange={e => setUpiTxnId(e.target.value)}
                  placeholder="e.g. 338491028475"
                  className="w-full bg-[#141414] border border-[#2b2b2b] rounded-xl px-3 py-2 text-xs text-white placeholder-gray-500 focus:outline-none focus:border-[#D4AF37]"
                />
              </div>
            </div>
          )}

          <div className="p-4 rounded-2xl bg-[#181818] border border-[#262626] space-y-2">
            <span className="text-[11px] font-bold text-gray-400 uppercase tracking-wider block">
              Order Summary ({cart.length} items)
            </span>
            <div className="space-y-1 max-h-32 overflow-y-auto pr-1">
              {cart.map(ci => (
                <div key={ci.cartItemId} className="flex justify-between text-xs text-gray-300">
                  <span className="truncate pr-2">
                    {ci.item.name} <span className="text-gray-400">×{ci.quantity}</span>
                  </span>
                  <span className="shrink-0 font-medium font-cinzel">₹{ci.item.price * ci.quantity}</span>
                </div>
              ))}
            </div>

            <div className="pt-2 border-t border-[#2a2a2a] space-y-1 text-xs">
              <div className="flex justify-between text-gray-400">
                <span>Subtotal</span>
                <span>₹{subtotal}</span>
              </div>
              <div className="flex justify-between text-gray-400">
                <span>Delivery Fee</span>
                <span>{effectiveDeliveryFee === 0 ? 'FREE' : `₹${effectiveDeliveryFee}`}</span>
              </div>
              <div className="flex justify-between text-sm font-bold text-white pt-1">
                <span>Total Amount</span>
                <span className="text-lg text-[#D4AF37] font-cinzel font-black">₹{effectiveTotal}</span>
              </div>
            </div>
          </div>
        </div>

        <div className="p-5 sm:p-6 bg-[#181818] border-t border-[#242424] space-y-3">
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            <button
              type="button"
              onClick={() => handlePlaceOrder(false)}
              disabled={isSubmitting}
              className="py-3.5 px-4 rounded-xl gold-button text-sm font-bold flex items-center justify-center gap-2 shadow-lg shadow-[#D4AF37]/20 active:scale-98 disabled:opacity-50 cursor-pointer"
            >
              <CreditCard className="w-4 h-4" />
              <span>{isSubmitting ? 'Placing Order...' : `Confirm & Place Order · ₹${effectiveTotal}`}</span>
            </button>

            <button
              type="button"
              onClick={() => handlePlaceOrder(true)}
              disabled={isSubmitting}
              className="py-3.5 px-4 rounded-xl bg-[#25D366] hover:bg-[#20bd5a] text-black text-sm font-bold flex items-center justify-center gap-2 shadow-lg transition-all active:scale-98 disabled:opacity-50 cursor-pointer"
            >
              <PhoneCall className="w-4 h-4 fill-black" />
              <span>Order on WhatsApp</span>
            </button>
          </div>

          <p className="text-[11px] text-gray-400 text-center">
            🔒 By placing your order, your food will be freshly prepared at AL ISHRA RESTAURANT, Sasaram.
          </p>
        </div>
      </div>
    </div>
  );
};
